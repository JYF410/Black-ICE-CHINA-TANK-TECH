				set_variable = { ROOT.selected_research_slot = 1 }

			}
			research_slot_overlay_2_click = {
				set_variable = { ROOT.selected_research_slot = 2 }
				
				set_variable = { ROOT.selected_research_slot_state = -1 }
			}
			research_slot_overlay_click = {
				set_variable = { ROOT.selected_research_slot = i }
				add_to_variable = { ROOT.selected_research_slot = 3 }
				
				set_variable = { ROOT.selected_research_slot_state = v }
			} 
			research_slot_overlay_right_click = {
				if = {
					limit = {
						ROOT = { has_variable = selected_research_slot }
					}
				 	ROOT = {
						clear_variable = selected_research_slot
					}
				}
				else = {
					ROOT = {
						meta_effect = {
							text = {
								goto_state = [ID]
							}
							ID = "[Prev.GetID]"
						}
					}
				}
			}
			research_slot_lost_overlay_right_click = {
				if = {
					limit = {
						ROOT = { has_variable = selected_research_slot }
					}
				 	ROOT = {
						clear_variable = selected_research_slot
						add_to_variable = { ui_research = 1 }
					}
				}
				else = {
					ROOT = {
						meta_effect = {
							text = {
								goto_state = [ID]
							}
							ID = "[Prev.GetID]"
						}
					}
				}
			}
		}
		
		dynamic_lists = {
			research_slot_list = {
				array = research_slots
				change_scope = yes
				#entry_container = "[GetResearchSlotEntry]"
				entry_container = "research_slot_container"
			}
		}

				check_variable = { research_rights@ROOT^num < 10 }

	BI_request_armor_tab = {
		context_type = player_context
		 
		window_name = "BI_request_armor_tab_container"	
		parent_window_name = "countrytechtreeview"
		
		visible = {
			NOT = {
				has_country_flag = GER_armor
				has_country_flag = ITA_armor
				has_country_flag = JAP_armor
				has_country_flag = ENG_armor
				has_country_flag = FRA_armor
				has_country_flag = USA_armor
				has_country_flag = SOV_armor
				has_country_flag = CZE_armor
				has_country_flag = POL_armor
				has_country_flag = HUN_armor
				has_country_flag = SWE_armor
				has_country_flag = CHI_armor
			}
		}
		
		properties = {
			flag = {
				image = "[This.GetFlag]"
			}
		}
		
		dynamic_lists = {
			request_armor_list = {
				array = request_armor_list
				change_scope = yes
				entry_container = "request_armor_entry"
			}
		}
		
		triggers = {    
			flag_click_enabled = {
				if = {
					limit = {
						NOT = { has_global_flag = allowarmorrequest }
					}
					ROOT = {
						OR = {
							tank_production > 29
							tank_assembly > 5
							custom_trigger_tooltip = {
								tooltip = HAS_PRODUCED_200_TANKS
								check_variable = { total_equipment_produced_armor > 199 }
							}
						}
					}
					OR = {
						has_opinion = {
							target = ROOT
							value > 49
						}
						is_licensing_any_to = ROOT
					}
					if = {
						limit = {
							has_government = communism
						}
						ROOT = { has_government = communism }
					}
					else_if = {
						limit = {
							democratic_govt = yes
						}
						ROOT = { democratic_govt = yes }
					}
				}
			}
			bg_visible = {
				check_variable = { request_armor_list^num > 0 }
			}
		}
		
		effects = {
			request_armor_click = {
				if = {
					limit = {
						check_variable = { request_armor_list^num = 0 }
					}
					add_to_array = { request_armor_list = GER }
					add_to_array = { request_armor_list = ITA }
					add_to_array = { request_armor_list = JAP }
					add_to_array = { request_armor_list = ENG }
					add_to_array = { request_armor_list = FRA }
					add_to_array = { request_armor_list = USA }
					add_to_array = { request_armor_list = SOV }
				}
				else = {
					clear_array = request_armor_list
				}
			}
			flag_click = {
				
				custom_effect_tooltip = generic_skip_one_line_tt
				country_event = blackiceevents.164
				custom_effect_tooltip = needs_approval
				event_option_tooltip = blackiceevents.164.a
			}
		}
	}
